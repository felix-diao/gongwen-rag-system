## 2. 后端部署

### 2.1 Docker 服务部署

#### 2.1.1 Docker Compose 配置

`docker-compose.yml`:

```yaml
version: '3.8'

services:
  # PostgreSQL 数据库
  postgres:
    image: postgres:17.2
    container_name: gongwen-postgres
    environment:
      POSTGRES_USER: gongwen
      POSTGRES_PASSWORD: password
      POSTGRES_DB: gongwenrag
    volumes:
      - postgres_data:/var/lib/postgresql/data
      - ./backups:/backups
    ports:
      - "5432:5432"
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U gongwen"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped

  # PostgreSQL 备份服务
  postgres-backup:
    image: prodrigestivill/postgres-backup-local:17
    container_name: gongwen-postgres-backup
    environment:
      POSTGRES_HOST: postgres
      POSTGRES_DB: gongwenrag
      POSTGRES_USER: gongwen
      POSTGRES_PASSWORD: password
      SCHEDULE: "@daily"
      BACKUP_KEEP_DAYS: 7
      BACKUP_KEEP_WEEKS: 2
      BACKUP_KEEP_MONTHS: 2
    volumes:
      - ./backups:/backups
    depends_on:
      - postgres
    restart: unless-stopped

  # etcd (Milvus 元数据存储)
  etcd:
    image: quay.io/coreos/etcd:v3.5.5
    container_name: gongwen-etcd
    environment:
      ETCD_AUTO_COMPACTION_MODE: revision
      ETCD_AUTO_COMPACTION_RETENTION: 1000
      ETCD_QUOTA_BACKEND_BYTES: 4294967296
    volumes:
      - etcd_data:/etcd
    command: etcd -advertise-client-urls=http://127.0.0.1:2379 -listen-client-urls=http://0.0.0.0:2379 --data-dir /etcd
    healthcheck:
      test: ["CMD", "etcdctl", "endpoint", "health"]
      interval: 30s
      timeout: 20s
      retries: 3
    restart: unless-stopped

  # MinIO (Milvus 存储后端)
  minio:
    image: minio/minio:RELEASE.2023-03-20T20-16-18Z
    container_name: gongwen-minio
    environment:
      MINIO_ACCESS_KEY: minioadmin
      MINIO_SECRET_KEY: minioadmin
    volumes:
      - minio_data:/minio_data
    command: minio server /minio_data
    ports:
      - "9000:9000"
      - "9001:9001"
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:9000/minio/health/live"]
      interval: 30s
      timeout: 20s
      retries: 3
    restart: unless-stopped

  # Milvus 向量数据库
  milvus:
    image: milvusdb/milvus:v2.6.4
    container_name: gongwen-milvus
    environment:
      ETCD_ENDPOINTS: etcd:2379
      MINIO_ADDRESS: minio:9000
    volumes:
      - milvus_data:/var/lib/milvus
    ports:
      - "19530:19530"
    depends_on:
      - etcd
      - minio
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:19530/v1/health"]
      interval: 30s
      timeout: 20s
      retries: 3
    restart: unless-stopped

  # Attu (Milvus 管理界面)
  attu:
    image: zilliz/attu:v2.2.2
    container_name: gongwen-attu
    environment:
      MILVUS_URL: milvus:19530
    ports:
      - "3000:3000"
    depends_on:
      - milvus
    restart: unless-stopped

  # RAG 服务
  rag-service:
    build:
      context: ./app
      dockerfile: Dockerfile
    container_name: gongwen-rag-service
    environment:
      - DATABASE_URL=postgresql://gongwen:password@postgres:5432/gongwenrag
      - MILVUS_HOST=milvus
      - MILVUS_PORT=19530
    volumes:
      - ./uploads:/app/uploads
      - ./generated_documents:/app/generated_documents
      - ./meeting_files:/app/meeting_files
      - ./logs:/app/logs
      - ./models:/app/models
    ports:
      - "8080:8080"
    depends_on:
      postgres:
        condition: service_healthy
      milvus:
        condition: service_healthy
    restart: unless-stopped

volumes:
  postgres_data:
  etcd_data:
  minio_data:
  milvus_data:
```

#### 2.2.4 初始化脚本说明

项目提供的初始化脚本：

| 脚本 | 用途 | 执行时机 |
|------|------|----------|
| init_db.py | 创建数据库表结构 | 首次部署 |
| init_admin.py | 创建管理员账户 | 首次部署 |

脚本执行方式：

```bash
# 本地执行
python scripts/init_db.py

# Docker 容器内执行
docker-compose exec rag-service python scripts/init_db.py

# 带参数执行
python scripts/import_documents.py --dir /path/to/documents --base-id 1
```

### 3.3 部署前端

#### 3.3.1 Nginx 配置

```nginx
server {
    listen 80;
    server_name your-domain.com;

    # 前端静态文件
    location / {
        root /var/www/document-writer/dist;
        try_files $uri $uri/ /index.html;
    }

    # API 代理
    location /api/ {
        proxy_pass http://rag-service:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_cache_bypass $http_upgrade;
    }

    # WebSocket 代理
    location /ws/ {
        proxy_pass http://rag-service:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_read_timeout 86400;
    }

    # HTTPS 配置
    listen 443 ssl;
    ssl_certificate /etc/nginx/ssl/cert.pem;
    ssl_certificate_key /etc/nginx/ssl/key.pem;
}
```

#### 3.3.2 部署静态文件

## 7. 备份恢复

### 7.1 数据备份

#### 7.1.1 PostgreSQL 备份

自动备份（已配置）：

```yaml
# docker-compose.yml 中已配置备份服务
postgres-backup:
  image: prodrigestivill/postgres-backup-local:17
  environment:
    SCHEDULE: "@daily"
    BACKUP_KEEP_DAYS: 7
```

手动备份：

```bash
# 备份整个数据库
docker-compose exec postgres pg_dump -U gongwen gongwenrag > backup_$(date +%Y%m%d).sql

# 备份到 Docker 容器
docker-compose exec postgres pg_dump -U gongwen gongwenrag -f /backups/manual_backup.sql
```

#### 7.1.2 Milvus 备份

```bash
# 使用 Milvus 备份工具
# 参考: https://milvus.io/docs/backup_and_restore.md
```

#### 7.1.3 文件备份

```bash
# 备份上传文件
tar -czf uploads_backup_$(date +%Y%m%d).tar.gz uploads/

# 备份配置文件
tar -czf config_backup_$(date +%Y%m%d).tar.gz .env app/config.py
```

### 7.2 数据恢复

#### 7.2.1 PostgreSQL 恢复

```bash
# 恢复数据库
docker-compose exec -T postgres psql -U gongwen gongwenrag < backup_20240115.sql

# 或使用备份目录
docker-compose exec postgres psql -U gongwen gongwenrag -f /backups/daily/20240115.sql
```

#### 7.2.2 文件恢复

```bash
# 恢复上传文件
tar -xzf uploads_backup_20240115.tar.gz

# 恢复配置文件
tar -xzf config_backup_20240115.tar.gz
```

### 7.3 完整恢复流程

```bash
# 1. 停止服务
docker-compose down

# 2. 恢复数据库
cat backup.sql | docker-compose exec -T postgres psql -U gongwen gongwenrag

# 3. 恢复文件
tar -xzf uploads_backup.tar.gz
tar -xzf config_backup.tar.gz

# 4. 启动服务
docker-compose up -d

# 5. 验证服务
curl http://localhost:8080/health
```
